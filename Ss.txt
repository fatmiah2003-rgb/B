                        new QRCode(qrContainer, {
                            text: item.url || 'https://example.com',
                            width: 78,
                            height: 78,
                            colorDark: "#3e2723",
                            colorLight: "#ffffff",
                            correctLevel: QRCode.CorrectLevel.M
                        });
                    }
                }, 10);
            });
        }

        // فتح نافذة تعديل عنصر واحد
        function openItemModal(id) {
            const item = portfolioItems.find(i => i.id === id);
            if (!item) return;

            currentEditingId = id;
            document.getElementById('modalBadge').textContent = item.id;
            document.getElementById('modalTitle').textContent = item.title;
            document.getElementById('modalUrlInput').value = item.url;
            document.getElementById('modalTextInput').value = item.title;
            document.getElementById('modalVisitLink').href = item.url;

            const previewBox = document.getElementById('modalQrPreview');
            previewBox.innerHTML = '';
            new QRCode(previewBox, {
                text: item.url || 'https://example.com',
                width: 140,
                height: 140,
                colorDark: "#3e2723",
                colorLight: "#ffffff"
            });

            const modal = document.getElementById('itemModal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeModal() {
            const modal = document.getElementById('itemModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
            currentEditingId = null;
        }

        function saveItemChanges() {
            if (!currentEditingId) return;
            const item = portfolioItems.find(i => i.id === currentEditingId);
            const newUrl = document.getElementById('modalUrlInput').value.trim();
            const newTitle = document.getElementById('modalTextInput').value.trim();

            if (newUrl) item.url = newUrl;
            if (newTitle) item.title = newTitle;

            saveData();
            renderCards();
            closeModal();
            showToast('تم تحديث العنصر ورمز الـ QR بنجاح ✨');
        }

        // تصدير التصميم كصورة عالية الجودة
        function downloadAsImage() {
            const btn = document.getElementById('downloadBtn');
            const originalText = btn.innerHTML;
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> <span>جاري التجهيز...</span>';
            btn.disabled = true;

            const captureElement = document.getElementById('capture-area');
            
            html2canvas(captureElement, {
                scale: 2.5,
                useCORS: true,
                backgroundColor: '#fdf8f5',
                logging: false
            }).then(canvas => {
                const link = document.createElement('a');
                const teacherName = document.getElementById('teacherNameInput').value.trim() || 'ملف_الأداء_الوظيفي';
                link.download = `${teacherName}.png`;
                link.href = canvas.toDataURL('image/png');
                link.click();

                btn.innerHTML = originalText;
                btn.disabled = false;
                showToast('تم تحميل الصورة بنجاح! 📸');
            }).catch(err => {
                console.error(err);
                btn.innerHTML = originalText;
                btn.disabled = false;
                showToast('حدث خطأ أثناء حفظ الصورة');
            });
        }

        function showToast(message) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.remove('opacity-0');
            setTimeout(() => {
                toast.classList.add('opacity-0');
            }, 3000);
        }
